cmake_minimum_required(VERSION 3.22)
project(clipboard-panel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets DBus Sql Test)
find_package(KF6GuiAddons REQUIRED)
find_package(KF6GlobalAccel REQUIRED)
add_library(panel-core history.cpp)
target_include_directories(panel-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(panel-core PUBLIC Qt6::Core Qt6::Gui Qt6::Sql)
add_library(panel-ui panel.cpp)
target_link_libraries(panel-ui PUBLIC panel-core Qt6::Widgets)
add_executable(clipboard-panel main.cpp kwin-bridge.cpp)
target_link_libraries(clipboard-panel PRIVATE panel-ui Qt6::DBus KF6::GuiAddons KF6::GlobalAccel)
install(TARGETS clipboard-panel RUNTIME DESTINATION libexec)
enable_testing()
add_executable(history-test tests/history-test.cpp)
target_link_libraries(history-test PRIVATE panel-core Qt6::Test)
add_test(NAME history COMMAND history-test)
add_executable(panel-test tests/panel-test.cpp)
target_link_libraries(panel-test PRIVATE panel-ui Qt6::Test)
add_test(NAME panel COMMAND panel-test)
set_tests_properties(panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME installer COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/installer-test.py)
