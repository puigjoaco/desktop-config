cmake_minimum_required(VERSION 3.22)
project(mattux VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets DBus Test)
find_package(KF6StatusNotifierItem 6.14 REQUIRED)
find_package(KF6GlobalAccel REQUIRED)
find_package(KF6XmlGui REQUIRED)
find_package(KF6Config REQUIRED)
find_package(KF6GuiAddons REQUIRED)
add_executable(mattux main.cpp tray.cpp tray.h logic.h icon.h shortcut-dialog.cpp shortcut-dialog.h)
target_link_libraries(mattux PRIVATE Qt6::Widgets Qt6::DBus KF6::StatusNotifierItem KF6::GlobalAccel KF6::XmlGui KF6::ConfigCore KF6::GuiAddons)
install(TARGETS mattux RUNTIME DESTINATION libexec)
enable_testing()
add_executable(logic-test tests/logic-test.cpp logic.h)
target_link_libraries(logic-test PRIVATE Qt6::Gui Qt6::Test)
add_test(NAME logic COMMAND logic-test)
add_test(NAME scripts COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/check-scripts.sh)

# Manual private-bus/nested-KWin UI check; never drive the user's desktop.
add_executable(shortcut-dialog-test EXCLUDE_FROM_ALL tests/shortcut-dialog-test.cpp shortcut-dialog.cpp shortcut-dialog.h tray.cpp tray.h)
target_link_libraries(shortcut-dialog-test PRIVATE Qt6::Widgets Qt6::DBus Qt6::Test KF6::GlobalAccel KF6::XmlGui KF6::GuiAddons KF6::StatusNotifierItem KF6::ConfigCore)
